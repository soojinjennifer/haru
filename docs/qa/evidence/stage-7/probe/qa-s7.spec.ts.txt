import { inflateSync } from 'node:zlib';
import { expect, test, type Page } from '@playwright/test';
import { hideDevPanel, settle } from '../../../../../e2e/support/gallery';
import { logInToHome, settleImages } from '../../../../../e2e/support/session';

const LATEST = 'e7e7e7e7-0000-4000-8000-000000000002';
const PDF_BTN = '.haru-report-pdf .haru-button';
type Rgb = { r: number; g: number; b: number };
function px(png: Buffer): Rgb {
  const chunks: Buffer[] = []; let o = 8;
  while (o + 8 <= png.length) { const l = png.readUInt32BE(o); const t = png.toString('ascii', o + 4, o + 8); if (t === 'IDAT') chunks.push(png.subarray(o + 8, o + 8 + l)); o += 12 + l; }
  const raw = inflateSync(Buffer.concat(chunks)); return { r: raw[1]!, g: raw[2]!, b: raw[3]! };
}
const at = async (p: Page, x: number, y: number) => px(await p.screenshot({ clip: { x, y, width: 1, height: 1 } }));
const d = (a: Rgb, b: Rgb) => Math.abs(a.r - b.r) + Math.abs(a.g - b.g) + Math.abs(a.b - b.b);
const log = (k: string, v: unknown) => console.log(`QA ${k} ${JSON.stringify(v)}`);

async function openDetail(page: Page) {
  await logInToHome(page);
  await page.evaluate((t) => { window.history.pushState({}, '', t); window.dispatchEvent(new PopStateEvent('popstate')); }, `/progress/reports/${LATEST}`);
  await expect(page.locator('.haru-report-detail__body[data-state="ready"]')).toBeVisible();
  await settleImages(page);
}

for (const width of [320, 393, 600]) {
  for (const run of [1, 2, 3]) {
    test(`F-09 detail ${width} right probe run${run}`, async ({ page }) => {
      await hideDevPanel(page);
      await page.setViewportSize({ width, height: width === 320 ? 568 : width === 393 ? 852 : 800 });
      await openDetail(page);
      const box = await page.locator(PDF_BTN).first().evaluate((n) => { n.scrollIntoView({ block: 'center' }); const r = n.getBoundingClientRect(); return { l: r.left, r: r.right, t: r.top, b: r.bottom, vw: window.innerWidth, scrollbarW: window.innerWidth - document.documentElement.clientWidth }; });
      await settle(page);
      const midY = Math.round((box.t + box.b) / 2);
      const pts = [1, 2, 3, 5, 8, 12, 20].map((k) => ({ k, x: Math.round(box.r) + k, y: midY })).filter((p) => p.x < box.vw);
      const info = await page.evaluate((ps) => ps.map((p) => { const e = document.elementFromPoint(p.x, p.y); return e ? `${e.tagName}.${(e.getAttribute('class') ?? '').split(' ')[0]}` : null; }), pts);
      const before: Rgb[] = []; for (const p of pts) before.push(await at(page, p.x, p.y));
      await page.addStyleTag({ content: `${PDF_BTN} { box-shadow: none !important; }` }); await settle(page);
      const res = []; for (const [i, p] of pts.entries()) { const a = await at(page, p.x, p.y); res.push({ k: p.k, delta: d(before[i]!, a), on: before[i], off: a, el: info[i] }); }
      log(`F09-${width}-run${run}`, { box: { l: box.l, r: box.r, vw: box.vw, sb: box.scrollbarW }, res });
    });
  }
}

test('detail 393 max-scroll bottom shadow', async ({ page }) => {
  await hideDevPanel(page);
  await openDetail(page);
  const info = await page.evaluate(() => {
    const btn = document.querySelector('.haru-report-pdf .haru-button')!;
    let sc: Element | null = btn.parentElement;
    while (sc && !/(auto|scroll)/.test(getComputedStyle(sc).overflowY)) sc = sc.parentElement;
    if (!sc) return null;
    sc.scrollTop = sc.scrollHeight;
    const r = btn.getBoundingClientRect(); const s = sc.getBoundingClientRect();
    const all = [...document.querySelectorAll('.haru-report-pdf .haru-button')].map((b) => { const q = b.getBoundingClientRect(); return { top: q.top, bottom: q.bottom }; });
    return { sc: sc.className, scBottom: s.bottom, vh: window.innerHeight, btnBottom: r.bottom, buttons: all, scrollTop: sc.scrollTop, max: sc.scrollHeight - sc.clientHeight, padBottom: getComputedStyle(sc).paddingBottom };
  });
  await settle(page);
  log('maxscroll', info);
  await page.screenshot({ path: '/Users/soojin/dev/haru/docs/qa/evidence/stage-7/qa-detail-393-maxscroll.png' });
  const last = info!.buttons[info!.buttons.length - 1]!;
  const ys = [2, 6, 10, 20, 30, 40, 60, 77].map((k) => Math.round(last.bottom) + k).filter((y) => y < info!.vh);
  const before: Rgb[] = []; for (const y of ys) before.push(await at(page, 196, y));
  await page.addStyleTag({ content: `${PDF_BTN} { box-shadow: none !important; }` }); await settle(page);
  const res = []; for (const [i, y] of ys.entries()) res.push({ y, below: y - Math.round(last.bottom), delta: d(before[i]!, await at(page, 196, y)) });
  // edge row (last visible row of scroll container) delta
  log('maxscroll-probe', res);
});

test('progress success a11y/text probes', async ({ page }) => {
  await hideDevPanel(page);
  await logInToHome(page);
  await page.getByRole('navigation').getByRole('link', { name: 'Progress' }).click();
  await expect(page.locator('.haru-report-card[data-state="published"]')).toBeVisible();
  await settleImages(page);
  const r = await page.evaluate(() => {
    const main = document.querySelector('main')!;
    const small = [...main.querySelectorAll('*')].filter((e) => [...e.childNodes].some((n) => n.nodeType === 3 && n.textContent!.trim()) && (e as HTMLElement).offsetParent !== null && parseFloat(getComputedStyle(e).fontSize) < 11).map((e) => e.textContent);
    return { progressbar: main.querySelectorAll('[role="progressbar"]').length, valuenow: main.querySelectorAll('[aria-valuenow]').length, ariaExpanded: main.querySelectorAll('[aria-expanded]').length, small, hscroll: document.documentElement.scrollWidth > window.innerWidth, text: main.innerText.slice(0, 1500) };
  });
  log('progress', r);
  expect(r.progressbar).toBe(0);
  expect(r.valuenow).toBe(0);
});
