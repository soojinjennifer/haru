import { test, expect, type Page } from '@playwright/test';
import { logInToHome, firstStartSession, settleImages } from '/Users/soojin/dev/haru/e2e/support/session';
import { gotoApp, hideDevPanel, settle } from '/Users/soojin/dev/haru/e2e/support/gallery';
import * as fs from 'node:fs';
const OUT = '/Users/soojin/dev/haru/docs/qa/evidence/stage-6';
const L = (s: string) => fs.appendFileSync(OUT + '/r3-probe.log', s + '\n');
async function spa(page: Page, path: string) {
  await page.evaluate((t) => { window.history.pushState({}, '', t); window.dispatchEvent(new PopStateEvent('popstate')); }, path);
}
async function dump(page: Page, tag: string) {
  const r = await page.evaluate(() => ({
    radio: Array.from(document.querySelectorAll('.haru-radio-group')).map((g) => g.className + ' :: ' + Array.from(g.querySelectorAll('.haru-radio-group__label')).map((l) => (l.textContent ?? '').trim().slice(0, 24) + '=' + getComputedStyle(l).fontWeight).join(' | ')),
    section: Array.from(document.querySelectorAll('.haru-text--tracking-section-label')).map((e) => { const c = getComputedStyle(e); return `${e.textContent}|w${c.fontWeight}|${c.fontSize}|ls${c.letterSpacing}|${c.color}`; }),
  }));
  L(`[${tag}] ` + JSON.stringify(r));
}
for (const [tag, path] of [['onb-growth', '/onboarding/growth-capability'], ['settings-growth', '/settings/growth-capability']] as const) {
  test(tag, async ({ page }) => {
    await hideDevPanel(page); await logInToHome(page); await spa(page, path);
    await expect(page.locator('.haru-growth-focus [data-state="ready"]')).toBeVisible(); await settle(page);
    await dump(page, tag); await page.screenshot({ path: OUT + `/r3-${tag}.png` });
  });
}
for (const [tag, path] of [['onb-profile', '/onboarding/profile'], ['settings-profile', '/settings/profile']] as const) {
  test(tag, async ({ page }) => {
    await hideDevPanel(page); await logInToHome(page); await spa(page, path);
    await expect(page.locator('.haru-profile-view [data-state="ready"]')).toBeVisible(); await settle(page);
    await dump(page, tag);
  });
}
test('signup', async ({ page }) => {
  await gotoApp(page, '/auth/sign-up'); await settle(page); await dump(page, 'signup');
});
test('reading context list variant', async ({ page }) => {
  await hideDevPanel(page); await logInToHome(page);
  await firstStartSession(page).click();
  const sheet = page.getByRole('dialog', { name: 'When will you read?' });
  await expect(sheet).toBeVisible();
  await sheet.getByText('Pick a time…', { exact: true }).click();
  await sheet.getByRole('button', { name: 'Continue' }).click().catch(() => {});
  await page.waitForTimeout(800); await page.mouse.move(0, 0);
  await dump(page, 'reading-context-after-pick');
  await page.screenshot({ path: OUT + '/r3-reading-context-step2.png' });
});
