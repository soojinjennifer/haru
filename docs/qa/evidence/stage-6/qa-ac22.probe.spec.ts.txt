import { test, expect } from '@playwright/test';
import { logInToHome, firstStartSession, settleImages } from '/Users/soojin/dev/haru/e2e/support/session';
import { hideDevPanel } from '/Users/soojin/dev/haru/e2e/support/gallery';
import * as fs from 'node:fs';
test('AC-6-22 cross-screen name', async ({ page }) => {
  await hideDevPanel(page);
  await logInToHome(page);
  await page.getByRole('button', { name: 'Settings' }).click();
  await page.getByRole('button', { name: 'Child info' }).click();
  const fn = page.getByRole('textbox', { name: /First name/ });
  await expect(fn).toHaveValue('Emma');
  await fn.fill('Mia');
  await page.getByRole('button', { name: 'Save', exact: true }).click();
  await expect(page).toHaveURL(/\/settings$/);
  await page.getByRole('button', { name: 'Back' }).click();
  const homeText = (await page.locator('main').innerText()).replace(/\s+/g, ' ');
  await firstStartSession(page).click();
  const sheet = page.getByRole('dialog', { name: 'When will you read?' });
  await sheet.getByText('Bedtime', { exact: true }).click();
  await sheet.getByRole('button', { name: 'Continue' }).click();
  await expect(page.getByRole('button', { name: 'Start recording' })).toBeVisible();
  await settleImages(page);
  const sessionText = (await page.locator('body').innerText()).replace(/\s+/g, ' ');
  await page.getByRole('button', { name: 'Start recording' }).click();
  await expect(page.getByRole('button', { name: 'Pause recording' })).toBeVisible();
  await page.getByRole('button', { name: 'Submit' }).click();
  await expect(page.getByRole('region', { name: 'How was storytime with Mia?' })).toBeVisible();
  await page.waitForTimeout(800);
  const completeText = (await page.locator('body').innerText()).replace(/\s+/g, ' ');
  await page.screenshot({ path: '/Users/soojin/dev/haru/docs/qa/evidence/stage-6/qa-ac22-complete-after-rename.png' });
  fs.writeFileSync('/Users/soojin/dev/haru/docs/qa/evidence/stage-6/qa-ac22.log', ['HOME: ' + homeText, 'SESSION: ' + sessionText, 'COMPLETE: ' + completeText].join('\n') + '\n');
});
