async function spa(page: Page, path: string) {
  await page.evaluate((t) => { window.history.pushState({}, '', t); window.dispatchEvent(new PopStateEvent('popstate')); }, path);
}

function lum(c: number[]) { const f = (v: number) => { v /= 255; return v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; }; return 0.2126 * f(c[0]) + 0.7152 * f(c[1]) + 0.0722 * f(c[2]); }
function ratio(a: number[], b: number[]) { const x = lum(a), y = lum(b); return (Math.max(x, y) + 0.05) / (Math.min(x, y) + 0.05); }
async function bgBehind(page: Page, sel: string) {
  // hide text, screenshot element, average pixels
  const el = page.locator(sel).first();
  await el.scrollIntoViewIfNeeded();
  const color = await el.evaluate((n) => getComputedStyle(n).color);
  await el.evaluate((n) => { (n as HTMLElement).style.setProperty('color', 'transparent', 'important'); });
  const buf = await el.screenshot();
  await el.evaluate((n) => { (n as HTMLElement).style.removeProperty('color'); });
  const { PNG } = await import('/Users/soojin/dev/haru/node_modules/.pnpm/node_modules/pngjs/lib/png.js').catch(() => ({ PNG: null as any }));
  return { color, buf, PNG };
}
