async function changeFront(page: Page) {
  const front = page.locator('.haru-swipe-deck__card--front');
  if (await front.getByRole('checkbox').count() > 0) {
    const cb = front.getByRole('checkbox').first(); await cb.focus(); await page.keyboard.press('Space');
    return 'checkbox0=' + (await cb.isChecked()) + ':' + (await cb.getAttribute('value'));
  }
  const checked = front.getByRole('radio', { checked: true });
  if (await checked.count() === 0) { await front.getByRole('radio').first().focus(); await page.keyboard.press('Space'); }
  else { await checked.focus(); await page.keyboard.press('ArrowDown'); }
  return front.getByRole('radio', { checked: true }).getAttribute('value');
}

test('QA-6-003 pagehide + TopBar leave', async ({ page }) => {
  await openSurveyCards(page);
  const n0 = (await requestPayloads(page, SAVE)).length;
  await page.evaluate(() => window.dispatchEvent(new Event('pagehide')));
  await page.waitForTimeout(300);
  const n1 = (await requestPayloads(page, SAVE)).length;
  await changeFront(page);
  await page.evaluate(() => window.dispatchEvent(new Event('pagehide')));
  await expect.poll(async () => (await requestPayloads(page, SAVE)).length).toBe(n1 + 1);
  const last = (await requestPayloads(page, SAVE)).at(-1) as any;
  L('[pagehide] noEdit PUTs=' + (n1 - n0) + ' afterEdit +1 payload step=' + last.step + ' currentStep=' + String(last.currentStep));
  // TopBar back on first card (edit already flagged unsaved remains true after pagehide)
  await page.getByRole('button', { name: 'Go back' }).click();
  await expect(page).toHaveURL(/\/settings\/child\/survey$/);
  await page.waitForTimeout(300);
  const n3 = (await requestPayloads(page, SAVE)).length;
  L('[TopBar leave] PUTs after Go back (from first card, unsaved) total=' + n3 + ' (prev ' + (n1 + 1) + ')');
});

test('G-3 contrast 852 & 1200', async ({ page }) => {
  for (const H of [852, 1200]) {
    await page.setViewportSize({ width: 393, height: H });
    for (const [label, path, scen, text] of [['onb-youcanchange', '/onboarding/growth-capability', undefined, 'You can change this anytime in Settings.'], ['onb-choose', '/onboarding/growth-capability', 'children:all-scores-missing', 'Choose one to continue.'], ['onb-youcanchange-empty', '/onboarding/growth-capability', 'children:all-scores-missing', 'You can change this anytime in Settings.'], ['set-choose', '/settings/growth-capability', 'children:all-scores-missing', 'Choose one to continue.']] as const) {
      await open(page, path, scen);
      const loc = page.getByText(text, { exact: true }).first();
      if (!(await loc.isVisible().catch(() => false))) { await page.waitForTimeout(1500); }
      if (await loc.count() === 0) { L('[contrast] ' + H + ' ' + label + ' NOT PRESENT'); continue; }
      await loc.scrollIntoViewIfNeeded(); await settle(page);
      const info = await loc.evaluate((el) => { const r = el.getBoundingClientRect(); return { color: getComputedStyle(el).color, fs: getComputedStyle(el).fontSize, fw: getComputedStyle(el).fontWeight, x: r.x, y: r.y, w: r.width, h: r.height }; });
      await loc.evaluate((el) => (el as HTMLElement).style.setProperty('color', 'transparent', 'important'));
      const shot = await page.screenshot({ clip: { x: info.x, y: info.y, width: info.w, height: info.h } });
      await loc.evaluate((el) => (el as HTMLElement).style.removeProperty('color'));
      const bmp = decodePng(shot); const fg = parseCssRgb(info.color);
      let min = 99, max = 0;
      for (let i = 0; i < bmp.width * bmp.height; i++) { const c = { r: bmp.data[i * 4], g: bmp.data[i * 4 + 1], b: bmp.data[i * 4 + 2] }; const k = contrastRatio(fg, c); if (k < min) min = k; if (k > max) max = k; }
      if (H === 852) fs.writeFileSync(OUT + '/r2-contrast-bg-' + label + '.png', shot);
      L('[contrast] H=' + H + ' ' + label + ' color=' + info.color + ' ' + info.fs + '/' + info.fw + ' y=' + Math.round(info.y) + ' min=' + min.toFixed(2) + ' max=' + max.toFixed(2));
    }
  }
});
