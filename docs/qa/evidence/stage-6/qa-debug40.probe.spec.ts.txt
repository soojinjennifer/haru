import { test, expect } from '@playwright/test';
import { logInToHome, requestOperations, requestPayloads } from '/Users/soojin/dev/haru/e2e/support/session';
test('debug 40', async ({ page }) => {
  const out: string[] = [];
  await logInToHome(page, 'activity-cards:difficulty-lowered');
  await page.locator('.haru-activity-card-flip__face--front').first().getByRole('button', { name: 'Start Session' }).click();
  const sheet = page.getByRole('dialog', { name: 'When will you read?' });
  await sheet.getByText('Bedtime', { exact: true }).click();
  await sheet.getByRole('button', { name: 'Continue' }).click();
  await expect(page.locator('.haru-scrim')).toHaveCount(0);
  await page.getByRole('button', { name: 'Start recording' }).click();
  await expect(page.getByRole('button', { name: 'Pause recording' })).toBeVisible();
  await page.getByRole('button', { name: 'Submit' }).click();
  await expect(page).toHaveURL(/\/complete$/);
  await page.getByRole('button', { name: 'Tricky today' }).click();
  await page.getByRole('button', { name: 'Save' }).click();
  await expect(page).toHaveURL(/\/$/);
  out.push('ops: ' + JSON.stringify(await requestOperations(page)));
  const refl = (await requestOperations(page)).filter(o => /reflection/i.test(o));
  for (const op of new Set(refl)) out.push(op + ' ' + JSON.stringify(await requestPayloads(page, op)));
  await page.waitForTimeout(31_000);
  await page.getByRole('link', { name: 'Library' }).click();
  await expect(page).toHaveURL(/library/);
  await page.getByRole('link', { name: 'Home' }).click();
  await expect(page).toHaveURL(/\/$/);
  await page.waitForTimeout(2000);
  out.push('ops after revisit: ' + JSON.stringify((await requestOperations(page)).slice(-4)));
  out.push('first card: ' + (await page.locator('.haru-activity-card-flip__face--front').first().innerText()).replace(/\s+/g,' ').slice(0,160));
  const sess = (await requestPayloads(page, 'POST /v1/sessions/:sessionId/transitions'));
  out.push('transitions: ' + JSON.stringify(sess));
  const fs = await import('node:fs'); fs.writeFileSync('/Users/soojin/dev/haru/docs/qa/evidence/stage-6/qa-ac40-debug-library.log', out.join('\n') + '\n');
});
