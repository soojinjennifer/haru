import sys
from PIL import Image
a=Image.open(sys.argv[1]).convert('RGB'); b=Image.open(sys.argv[2]).convert('RGB')
th=int(sys.argv[3]) if len(sys.argv)>3 else 5
print('sizes',a.size,b.size)
W=min(a.size[0],b.size[0]);H=min(a.size[1],b.size[1])
pa=a.load();pb=b.load()
mask=[[False]*W for _ in range(H)];n=0
for y in range(H):
  for x in range(W):
    p=pa[x,y];q=pb[x,y]
    if max(abs(p[i]-q[i]) for i in range(3))>th: mask[y][x]=True;n+=1
print('diff px',n)
seen=[[False]*W for _ in range(H)];comps=[]
for y in range(H):
  for x in range(W):
    if mask[y][x] and not seen[y][x]:
      st=[(x,y)];seen[y][x]=True;c=0;x0=x1=x;y0=y1=y
      while st:
        cx,cy=st.pop();c+=1;x0=min(x0,cx);x1=max(x1,cx);y0=min(y0,cy);y1=max(y1,cy)
        for dx in(-1,0,1):
          for dy in(-1,0,1):
            nx,ny=cx+dx,cy+dy
            if 0<=nx<W and 0<=ny<H and mask[ny][nx] and not seen[ny][nx]: seen[ny][nx]=True;st.append((nx,ny))
      comps.append((c,(x0,y0,x1,y1)))
comps.sort(reverse=True)
print('components',len(comps))
for c in comps[:15]: print(c)
