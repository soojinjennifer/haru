async function spa(page: Page, path: string) {
  await page.evaluate((t) => { window.history.pushState({}, '', t); window.dispatchEvent(new PopStateEvent('popstate')); }, path);
}
async function measureChildForm(page: Page, tag: string) {
  const r = await page.evaluate(() => {
    const cs = (e: Element) => getComputedStyle(e);
    const card = document.querySelector('.haru-language-card') as HTMLElement | null;
    const out: Record<string, unknown> = {};
    if (card) {
      const c = cs(card);
      out.card = { radius: c.borderTopLeftRadius + '/' + c.borderTopRightRadius + '/' + c.borderBottomLeftRadius + '/' + c.borderBottomRightRadius,
        backdrop: c.backdropFilter, webkitBackdrop: (c as any).webkitBackdropFilter, bg: c.backgroundColor, shadow: c.boxShadow, rect: card.getBoundingClientRect().toJSON() };
      // any ancestor/descendant blur within the card subtree
      out.cardSubtreeBlur = [card, ...Array.from(card.querySelectorAll('*'))].filter((e) => { const f = cs(e).backdropFilter; return f && f !== 'none'; }).length;
    }
    const routine = document.querySelector('.haru-child-routine') as HTMLElement | null;
    if (routine) {
      const legend = routine.querySelector('.haru-radio-group__legend')!;
      const lc = cs(legend);
      const rows = Array.from(routine.querySelectorAll('.haru-radio-group__options > *'));
      const labels = Array.from(routine.querySelectorAll('.haru-radio-group__label'));
      out.routine = { groupClass: routine.querySelector('.haru-radio-group')?.className,
        legend: { text: legend.textContent, weight: lc.fontWeight, size: lc.fontSize, ls: lc.letterSpacing, color: lc.color, tt: lc.textTransform, rect: legend.getBoundingClientRect().toJSON() },
        labelWeights: labels.map((l) => (l.textContent ?? '').trim() + '=' + cs(l).fontWeight),
        firstRowTop: rows[0]?.getBoundingClientRect().top, lastRowBottom: rows.at(-1)?.getBoundingClientRect().bottom };
      if (card && rows[0]) out.gapCardToFirstRow = rows[0].getBoundingClientRect().top - card.getBoundingClientRect().bottom;
    }
    out.sectionLabels = Array.from(document.querySelectorAll('.haru-text--tracking-section-label')).map((e) => { const c = cs(e); return `${e.textContent}|w${c.fontWeight}|${c.fontSize}|ls${c.letterSpacing}|${c.color}`; });
    return out;
  });
  L(`[${tag}] ` + JSON.stringify(r));
  return r as any;
}

async function radioWeights(page: Page, tag: string) {
  const r = await page.evaluate(() => Array.from(document.querySelectorAll('.haru-radio-group')).map((g) => ({ cls: g.className,
    labels: Array.from(g.querySelectorAll('.haru-radio-group__label')).map((l) => (l.textContent ?? '').trim().slice(0, 40) + '=' + getComputedStyle(l).fontWeight + '/' + getComputedStyle(l).fontSize) })));
  L(`[${tag}] ` + JSON.stringify(r));
  return r;
}
